guard_usb() {
  route=$(ip -j route get 10.11.99.1)
  usb_interface=$(printf '%s' "$route" | jq -er 'if length == 1 and .[0].gateway == null then .[0].dev else error("not direct USB route") end')
  usb_address=$(printf '%s' "$route" | jq -er '.[0].prefsrc')
  case "$usb_interface" in ''|*[!A-Za-z0-9_.:-]*) exit 1 ;; esac
  case "$usb_address" in 10.11.99.[23456]) ;; *) printf '%s\n' 'ABORT: no approved USB IPv4 source'; exit 1 ;; esac
  test ! -e "/sys/class/net/$usb_interface/wireless"
  test ! -e "/sys/class/net/$usb_interface/phy80211"
  usb_device=$(readlink -f "/sys/class/net/$usb_interface/device")
  case "$usb_device" in */usb[0-9]*/*) ;; *) printf '%s\n' 'ABORT: route is not a physical USB network device'; exit 1 ;; esac
}
